"""Availability-aware Pangu selection. All stored timestamps are UTC Unix seconds."""
from __future__ import annotations

from datetime import datetime, timezone
from dataclasses import dataclass
from pathlib import Path
from zoneinfo import ZoneInfo
import numpy as np

HOUR = 3600
LATENCY = 12 * HOUR
PERIOD = 12 * HOUR


@dataclass(frozen=True)
class AvailabilityProtocol:
    name: str
    latency_hours: int
    period_hours: int
    maximum_lead_hours: int

    @property
    def latency(self):
        return self.latency_hours * HOUR

    @property
    def period(self):
        return self.period_hours * HOUR


TWICE_DAILY = AvailabilityProtocol('twice_daily_08_20_BJT_latency12h', 12, 12, 30)
DAILY = AvailabilityProtocol('daily_08_BJT_latency15h', 15, 24, 45)
PROTOCOLS = {p.name: p for p in (TWICE_DAILY, DAILY)}
LOCAL = ZoneInfo('Asia/Shanghai')
BAD_START = int(datetime(2024, 11, 1, tzinfo=LOCAL).timestamp())
BAD_END = int(datetime(2024, 12, 1, tzinfo=LOCAL).timestamp())
BAD_UTC_START = int(datetime(2024, 11, 1, tzinfo=timezone.utc).timestamp())
BAD_UTC_END = int(datetime(2024, 12, 1, tzinfo=timezone.utc).timestamp())


def unix(value: str, timezone_name: str = 'Asia/Shanghai') -> int:
    """Localize naive power timestamps; preserve timezone-aware NWP timestamps."""
    stamp = datetime.fromisoformat(value)
    if stamp.tzinfo is None:
        stamp = stamp.replace(tzinfo=ZoneInfo(timezone_name))
    return int(stamp.timestamp())


def select_issue(origin, protocol=TWICE_DAILY):
    origin = np.asarray(origin, dtype=np.int64)
    return ((origin - protocol.latency) // protocol.period) * protocol.period


def plan_times(origin, protocol=TWICE_DAILY):
    origin = np.asarray(origin, dtype=np.int64)
    if np.any(origin % 900):
        raise ValueError('Power origins must lie on the 15-minute time grid')
    issue = select_issue(origin, protocol)
    base = (origin // HOUR) * HOUR
    future = base[:, None] + np.arange(8, dtype=np.int64)[None] * HOUR
    past = base[:, None] + np.arange(-6, 0, dtype=np.int64)[None] * HOUR
    available = issue + protocol.latency
    assert np.all(available <= origin) and np.all(origin < available + protocol.period)
    assert np.all(issue % protocol.period == 0)
    assert np.all(future[:, 0] <= origin + 900)
    assert np.all(future[:, -1] >= origin + 6 * HOUR)
    assert np.all((future - issue[:, None]) <= protocol.maximum_lead_hours * HOUR)
    return dict(issue=issue, available=available, base=base, future=future, past=past)


def november_exclusions(origin, protocol=TWICE_DAILY):
    """Conservative exclusion of local power/weather windows and bad issue files.

    Covers the UTC filename month as well as the business calendar month. This
    also prevents December boundary samples from reusing a November issue.
    """
    origin = np.asarray(origin, dtype=np.int64)
    p = plan_times(origin, protocol)
    overlaps = lambda a, b: (a < BAD_END) & (b >= BAD_START)
    return {
        'power_window_touches_local_november': overlaps(origin - 23 * 900, origin + 24 * 900),
        'weather_window_touches_local_november': overlaps(p['past'][:, 0], p['future'][:, -1]),
        'issue_in_utc_november': (p['issue'] >= BAD_UTC_START) & (p['issue'] < BAD_UTC_END),
        'issue_in_local_november': (p['issue'] >= BAD_START) & (p['issue'] < BAD_END),
    }


def source_path(root: str | Path, issue: int, lead: int) -> Path:
    if lead < 0 or lead > 96:
        raise ValueError(f'Unsupported Pangu lead: {lead}')
    stamp = datetime.fromtimestamp(int(issue), tz=timezone.utc)
    step = f'{int(lead):02d}'
    return Path(root) / str(stamp.year) / step / f'{stamp:%Y-%m-%d %H:%M:%S}+00:00-{step}.npy'


def interpolate_nodes(nodes, offsets, queries):
    """Hourly linear interpolation without silent endpoint extrapolation."""
    nodes = np.asarray(nodes)
    offsets = np.asarray(offsets)
    queries = np.asarray(queries, dtype=np.float32)
    if queries.ndim == 1:
        queries = np.broadcast_to(queries, (len(nodes), len(queries)))
    if not np.allclose(np.diff(offsets, axis=1), 1):
        raise ValueError('Expected contiguous hourly nodes')
    x = queries - offsets[:, :1]
    if np.any(x < -1e-6) or np.any(x > nodes.shape[1] - 1 + 1e-6):
        raise ValueError('Required forecast valid time is outside available nodes')
    lo = np.minimum(np.floor(x).astype(np.int64), nodes.shape[1] - 2)
    frac = (x - lo)[..., None]
    rows = np.arange(len(nodes))[:, None]
    return (nodes[rows, lo] * (1 - frac) + nodes[rows, lo + 1] * frac).astype(np.float32)

    total = np.zeros(14, np.float64)
    square = np.zeros(14, np.float64)
    count = 0
    for start in range(0, len(ids), 128):
        x = np.asarray(raw[ids[start:start + 128]], dtype=np.float64)
        total += x.sum((0, 2, 3))
        square += np.square(x).sum((0, 2, 3))
        count += x.shape[0] * x.shape[2] * x.shape[3]
    mean = total / count
    std = np.sqrt(np.maximum(square / count - mean ** 2, 0))
    assert np.all(std > 1e-6)
    static = out / 'cache/static'
    static.mkdir(parents=True, exist_ok=True)
    for path in (OLD / 'static').glob('*.npy'):
        if path.stem not in ['nwp_mean', 'nwp_std']:
            shutil.copy2(path, static / path.name)
    mean, std = mean.astype('f4'), std.astype('f4')
    save(static / 'nwp_mean.npy', mean)
    save(static / 'nwp_std.npy', std)
    save(out / 'normalization_frame_index.npy', ids)
    atomic_json(out / 'NORMALIZATION.json', dict(fitted_split='train', frame_count=len(ids),
        value_count_per_channel=count, method='unique train-referenced physical crops, six past plus eight weather nodes',
        mean=mean.tolist(), std=std.tolist(), validation_or_test_used=False))
    return mean, std


def bilinear(values, ids, weights):
    # [..., channel, latitude, longitude] -> [..., station, channel]
    flat = values.reshape(*values.shape[:-2], -1)
    v = flat[..., ids]
    return (v * weights).sum(-1).swapaxes(-1, -2)


def dynamics(raw, ids, weights, station_lat):
    lat = np.load(OLD / 'static/weather_latitude.npy')
    lon = np.load(OLD / 'static/weather_longitude.npy')
    dphi = np.deg2rad(float(lat[1] - lat[0])) * 6371000
    dlam = np.deg2rad(float(lon[1] - lon[0])) * 6371000
    w = np.asarray(raw, dtype=np.float64)
    def point(field):
        return (field.reshape(*field.shape[:-2], -1)[..., ids] * weights).sum(-1)
    phi = np.deg2rad(station_lat)
    f = 2 * 7.2921159e-5 * np.sin(phi)
    ug = -point(np.gradient(w[:, :, 5], dphi, axis=-2, edge_order=2)) / f
    vg = point(np.gradient(w[:, :, 5], dlam, axis=-1, edge_order=2)) / np.cos(phi) / f
    dtdy = point(np.gradient(w[:, :, 9], dphi, axis=-2, edge_order=2))
    dtdx = point(np.gradient(w[:, :, 9], dlam, axis=-1, edge_order=2)) / np.cos(phi)
    p = bilinear(w, ids, weights)
    adv = -(p[..., 11] * dtdx + p[..., 13] * dtdy) * 3600
    dpdt = np.gradient(p[..., 0], 1., axis=1) / 100
    dvdt = np.gradient(np.hypot(p[..., 1], p[..., 2]), 1., axis=1)
    return np.stack([ug / 10, vg / 10, (p[..., 1] - ug) / 10, (p[..., 2] - vg) / 10,
                     adv, dpdt, dvdt], axis=-1).transpose(0, 2, 1, 3).astype('f4')
